if(first())
{
    Tick = 0
    MaxTick = 10
    
    # put the name of a player you want to target
    # if player is valid then e2 will become invisible
    Target = findPlayerByName("")
    
    if(Target:isValid())
    {
        entity():propDraw(0)
        entity():propNotSolid(1)
        entity():propNoDupe(1)
    }
    
    entity():propFreeze(1)
    ID = randint(10)
    Sounds = array(

        "vo/npc/female01/pain07.wav",
        "vo/npc/female01/pain08.wav",
        "vo/npc/female01/pain09.wav"
        
    )
    
    function playSound()
    {
        if(!soundPlaying(ID))
        {
            const Sound = Sounds[random(ID,Sounds:count()+1),string]
            
            entity():soundPlay(ID,1,Sound)
            soundPitch(ID,random(50,120))
            
        }
    }
}

    if(Target:isValid() & Player == Target)

        selfDestruct()

}

event tick()
{
    Tick++
    if(Tick > MaxTick)
    {
        Tick = 0
        MaxTick = randint(20)
    }
    if(Tick){exit()}
    
    
    playSound()
    
    if(Target:isValid())
    {
        TargetPos = Target:shootPos() + Target:eye() * (-50)
        entity():setPos(TargetPos)
    }
    
}
