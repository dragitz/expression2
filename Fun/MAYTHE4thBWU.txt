@name MAYTHE4THBW/U

@outputs Velocity
@persist [Forcer Me]:entity RequestedVel Forcer2:wirelink

#if(dupefinished() | duped()){ reset() }


Me = findPlayerByName("manners")
if(!Me:isValid()){ Me = owner() }

RequestedVel = 0

Forcer = entity():isWeldedTo()
if(!Forcer:isValid()){hint("Spawn a forcer first!",3.5), selfDestructAll()}
Forcer:createWire(entity(),"Velocity","Velocity")


Velocity = 0
hint("Press   0   to make the forcer invisible",3.5)

entity():propFreeze(1)

event tick()
{    
    if(Forcer:isValid() & Me:weapon():type() == "weapon_physcannon" & ->Velocity)
    {
        Forcer:setAlpha(0)
        Forcer:propNotSolid(1)
        Forcer:propFreeze(1)
        entity():propNotSolid(1)
        entity():setAlpha(0)
        
        Distance = 50
        Forward = Me:shootPos()+Me:eye()*Distance
        
        if(Forward:isInWorld() & owner():isAlive())
        {
            Forcer:setPos(Forward)
            
            Pitch = Me:eyeAngles():pitch() + 90
            Yaw = Me:eyeAngles():yaw()
            Roll = Me:eyeAngles():roll()
            
            Forcer:setAng(ang(Pitch,Yaw,Roll))            
        }
        
        Velocity = RequestedVel
        Forcer2["Velocity", number] = RequestedVel
        Forcer2["OffsetForce", number] = RequestedVel
        Forcer2["Force", number] = RequestedVel
    
    }
    
}

event keyPressed(Player:entity, Key:string, Down:number, _:string) {
    
    if(Player == Me)
    {
        if(Down == 1 & Player:weapon():type() == "weapon_physcannon") {
            
            switch (Key) {
                case "mouse_left",
                    RequestedVel = 10000
                    break
                case "mouse_right",
                    RequestedVel = -10000
                    break                
                case "0",
                    Forcer:setAlpha(0)
                    Forcer:propNotSolid(1)
                    Forcer:propFreeze(1)
                    break
                default,
                    RequestedVel = 0
            }
            
        }else{RequestedVel = 0}
    }
}
