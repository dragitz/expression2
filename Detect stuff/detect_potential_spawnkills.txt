MIN_SCORE = 2
if(first())
{
    
    Time = systime()
    Players = players()
    
    for(I=1,Players:count())
    {
        Key = Players[I,entity]:id()
        AliveTable[Key,number] = Time
        SpawnKilledTable[Key,number] = 0
    }
}
event playerConnected(Player:entity) {
    AliveTable[Player:id(),number] = systime()
    SpawnKilledTable[Key,number] = 0
}

event playerDeath(Victim:entity, Inflictor:entity, Attacker:entity) {
    # Check the player's alive time
    Time = systime()
    Key = Victim:id()
    TimeDiff = Time - AliveTable[Key,number]
    
    
    # Detect potential spawn kill
    if(TimeDiff < 1)
    {
        # Increase score
        SpawnKilledTable[Key,number] = SpawnKilledTable[Key,number] + 1
        
        # Check if vistim has died at least x times
        if(SpawnKilledTable[Key,number] >= MIN_SCORE)
        {
            local Model = Inflictor:model()
            local TrueAttacker = Inflictor:owner()
            print(Victim:name() + " is being spawnkilled by > ", TrueAttacker:name() + "    model: "+Model)
        }
        
    }else
    {
        # Victim has not been spawnkilled, decrease score by 1
        if(SpawnKilledTable[Key,number] > 0){ SpawnKilledTable[Key,number] = SpawnKilledTable[Key,number] - 1}
    }
    
    #print(TimeDiff, "   Score: ", SpawnKilledTable[Key,number])
     
    
}

event playerSpawn(Player:entity) {
    AliveTable[Player:id(),number] = systime()
}
