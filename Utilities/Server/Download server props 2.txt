    # Feel free to add more, but remember to implement it in the save n' load function
    function setEntityData(E:entity)
    {
        #local Pos = E:pos()
        if(E:model() == ""){ exit() }
        EntityData[E:id(),table] = table(
            "Model" = E:model(),
            "Pos"   = E:pos(),
            "Ang"   = E:angles(),
            "Mass"   = E:mass(),
            
            "Color"   = E:getColor(),
            "Alpha"   = E:getAlpha(),
            "Material"   = E:getMaterial(),
            
            "FrozenState" = E:isFrozen()
        )
        
    }    
    #  ------------------------------------------
    
    
    #  ------------------------------------------
    # This function will hold the table that will be stored in the new e2 file
    # NOTE: this has never crashed the chip. so I never bothered porting it to event tick()
    function constructFinalTable()
    {
        foreach(EntityIndex:number, EntId:number = Cache:keys()) {
            FinalTable["Entities",table][EntityIndex,number] = EntityIndex
            
            FinalTable["Data",table][EntityIndex,table] = EntityData[EntId,table]
            
            FinalTable["Welds",table][EntityIndex,array] = array()
            
            for(I=1,WeldTable[EntId,array]:count())
            {
                #print(EntId + " attached to ---> "+WeldTable[EntId,array][I,number])
                
                const EntityToReverse = WeldTable[EntId,array][I,number]
                #print(EntityToReverse)
                #const ReverseId = Cache[WeldTable[EntId,array][I,number],number]
                #const ReverseId = Cache["ReverseIndex",table][EntId,number]
                
                const ReverseId = Cache[EntityToReverse,number]
                
                #print(ReverseId + "( attached to ---> )"+EntId)
                
                FinalTable["Welds",table][EntityIndex,array]:pushNumber(ReverseId)
            }
                
        }
        
        # Write initial file here:
        if(!DISABLE_FILE_SAVE)
        {
            fileWrite(">e2shared/"+Final_name+".txt",FILE_BEGIN)
        }
        local MSG = "4/4 FinalTable constructed! generating code...", print(MSG), hint(MSG,3.5), Phase = 4, Tick = 1, exit()
    }

event tick()
{
    if(!SetupDone | !Phase){ exit() }
    
    #[
        Phase 1 = Cache props
        Phase 2 = computeConnections
        Phase 3 = constructFinalTable
        Phase 4 = generate code
        
    ]#
    
    Tick += 1
    if(Tick > MaxTick){ Tick = 0}
    
    if(perf(50) & !Tick)
    {
        
        if(Phase == 1)
        {
            if(CurrentCacheIndex > AllConnected:count() + 1){
                local MSG = "2/4 Cashing done! Computing connections...", print(MSG), hint(MSG,3.5), Phase = 2, exit()
                
                CachedKeys = Cache:keys() # this is meta caching
                exit()
            }
            
            # I wanted to make use of the function declared in first() but tick quota is reached,
            # this way is better overall..
            for(_=1,PropProcessPerTick)
            {
                local Ent = AllConnected[CurrentCacheIndex,entity]
                local EntityIndex = CurrentCacheIndex
                
                # bastard got through check
                if(!Ent:isValid() | !Lookup[Ent:getClass(),number]){ CurrentCacheIndex += 1, continue }
                
                Cache[Ent:id(), number] = EntityIndex
                ReverseIndex[EntityIndex,number] = Ent:id()
                
                #print(Ent:id() + " has ReverseIndex value of: "+EntityIndex)
                
                WeldTable[Ent:id(),array] = array()
                setEntityData(Ent)
                
                TotalProps += 1
                CurrentCacheIndex += 1
            }
            
            exit()
                       
        }
        elseif( Phase == 2)
        {
            if(Cache:keys():count() > 0 & CachedKeys:count() != Cache:keys():count())
            {
                CachedKeys = Cache:keys()
                #print("Yup, stored keys again: ",CachedKeys:count())
                exit()
            }
            
            for(_=1,WeldProcessPerTick)
            {
                # Find all weldings
                EntId = CachedKeys[CurrentWeldIndex,number]
                
                #const ReverseID = Cache[EntId, number]
                const Entity = entity(EntId)
                
                # Gotta be honest, this shouldn't happen...
                if(!Entity:isValid() | !Lookup[Entity:getClass(),number]){ CurrentWeldIndex += 1 }
                
                
                const Constraints = Entity:hasConstraints("weld")
                
                for(I=1,Constraints)
                {
                    # Returns the Nth entity E was constrained to with the given constraint type
                    local TargetEnt = Entity:isConstrainedTo("weld",I)
                    
                    if(!Entity:isValid() | !TargetEnt:isValid() | !Lookup[TargetEnt:getClass(),number]){ continue }
                    
                    #print("[+]   ",TargetEnt)
                    WeldTable[Entity:id(),array]:pushNumber(TargetEnt:id())
                }
              
                # -----------------------            
                CurrentWeldIndex += 1
                if(CurrentWeldIndex > AllConnected:count()){
                    local MSG = "3/4 Got welding data! constructing FinalTable...", print(MSG), hint(MSG,2.5), Phase = 3, exit()
                }
            }
            exit()
            
        }elseif( Phase == 3)
        {
            #FILE_BEGIN = "@name Spawning: "+Filename+"\n@persist CurrentIndex SetupDone FinalTable:table TotalEntities Phase\n\n#[\n    Phase 1 = Spawn\n    Phase 2 = Apply welds\n    Phase 3 = Unfreeze if necessary\n]#\n\nif(first())\n{\n    SetupDone = 0\n    CurrentIndex = 1\n    \n    Phase = 0\n    \n    FinalTable = table(\n        \"Entities\" = table(),\n        \"Data\" = table(),\n        \"Welds\" = table()\n    )\n    \n    TotalEntities = "+TotalProps+"\n\n"
            FILE_BEGIN = "@name Spawning: "+Filename+"\n@persist CurrentIndex SetupDone FinalTable:table TotalEntities Phase\n@persist Tick MaxTick\n\n@persist Errors MaxErrors\n@persist PropsTablePerTick\n\n@persist CurrentSetupIndex\n\n#[\n    Phase 1 = Spawn\n    Phase 2 = Apply welds\n    Phase 3 = Unfreeze if necessary\n]#\n\nif(first())\n{\n    TotalEntities = "+TotalProps+"\n    \n    SetupDone = 0\n    CurrentIndex = 1\n    \n    Phase = 0\n    \n    Tick = 0\n    MaxTick = 2\n    \n    Errors = 0\n    MaxErrors = 4\n    \n    PropsTablePerTick = 5\n    \n    FinalTable = table(\n        \"Entities\" = table(),\n        \"Data\" = table(),\n        \"Welds\" = table()\n    )\n    \n    CurrentSetupIndex = 1\n    \n    \n    print(\"Setting up table..\")\n}\n\nevent tick()\n{\n    if(!SetupDone | !Phase){\n    \n        Tick += 1\n        if(Tick > MaxTick){ Tick = 0}\n        \n        if(!Tick)\n        {\n            for(_=1,PropsTablePerTick)\n            {\n                switch(CurrentSetupIndex)\n                {\n"
            constructFinalTable()
            exit()
        }
        elseif( Phase == 4)
        {
            #[
            EncodedTable = vonEncode(FinalTable)
            CurrentCode = "FinalTable = vonDecodeTable("+EncodedTable+")\n"
            CurrentCode += FILE_END
            
            fileAppend(">e2shared/"+Final_name+".txt",CurrentCode)
            ]#
            
            if(!FinalTable["Entities",table][CurrentTableIndex,number])
            {
                # Check for remaining lines to be appended
                if(fileCanWrite())
                {
                    # Finish writing
                    if(Lines > 0)
                    {
                        if(!DISABLE_FILE_SAVE){
                            if(fileAppend(">e2shared/"+Final_name+".txt",CurrentCode))
                            {
                                CurrentCode = ""
                                Lines = 0
                            }
                        }
                        
                        exit()
                    }
                    
                    # Generate code
                    if(!DISABLE_FILE_SAVE){
                        fileAppend(">e2shared/"+Final_name+".txt",FILE_END)
                    }
                    local MSG = "Done! Check your: e2shared folder :D", print(MSG), hint(MSG,6.0)
                    
                    print("Time taken: ", systime() - InitialTime)
                    selfDestruct()
                    exit()
                }
                
                
            }
            
            for(_=1,LinesProcessPerTick)
            {
                CurrentCode += tableIndexToString(CurrentTableIndex)
    
                Lines += 1            
                CurrentTableIndex += 1
    
                if(Lines >= MaxLines & fileCanWrite())
                {
                    if(!DISABLE_FILE_SAVE){
                        fileAppend(">e2shared/"+Final_name+".txt",CurrentCode)
                    }
                    CurrentCode = ""
                    Lines = 0
                }
            }
            
        }
        
    }
}

event keyPressed(Player:entity, Key:string, Down:number, KeyBind:string) {
    if(SetupDone | !Down | Player != owner()){ exit() }
    
    if(KeyBind != "attack2" | owner():weapon():getClass() != "weapon_crowbar"){ exit() }
    
    const AimedEnt = owner():aimEntity()
    #if(AimedEnt:isValid() & AimedEnt:owner():isValid() & AimedEnt:owner() != owner() & AimedEnt:owner() != world() ){
    if(AimedEnt:isValid() & AimedEnt:owner():isValid() & AimedEnt:owner() != world() ){
        
        
        if(InitialEntity != AimedEnt & !ConfirmState)
        {
            InitialEntity = AimedEnt
            ConfirmState += 1
            
            local MSG = "Right click again to confirm."
            print(MSG)
            hint(MSG,3.5)
            exit()
        }
        elseif(InitialEntity:isValid() & InitialEntity == AimedEnt & ConfirmState)
        {
            AllConnected = InitialEntity:getConnectedEntities("weld")
            
            # This should add a bit more props
            findIncludePlayerProps(InitialEntity:owner())
            findByClass("prop_physics")
            findInSphere(InitialEntity:pos(),900)
            Extra = findToArray()
            for(I=1,Extra:count())
            {
                if(!Lookup[Extra[I,entity]:getClass(),number]){ continue }
                AllConnected:pushEntity(Extra[I,entity])
            }
            
            InitialTime = systime()
            
            local MSG = "1/4 Cashing props..."
            print(MSG)
            hint(MSG,3.5)
            
            SetupDone = 1
            Phase = 1
            exit()
        }
        
    }else
    {
        local MSG = "Invalid entity! Aim at something else!"
        print(MSG)
        hint(MSG,3.5)
        exit()
    }
}
