event chat(Player:entity, Message:string, Team:number) {
    
    if(Player == owner() && Message[1] == "!")
    {
        hideChat(1)
        Exploded = Message:explode(" ")
        
        Command = Exploded[1,string]
        Argument = Exploded[2,string]
        
        
        switch(Command)
        {
            case "!jj",
                print("New height: ",ReverseHeight,">>",Argument:toNumber())
                ReverseHeight = Argument:toNumber()
                
            break
            case "!sethome",
                HOME = owner():pos()
                print("Home set")
            break
            case "!home",
                if(HOME != vec(0,0,0))
                {
                    if(owner():isAlive())
                    {
                        goto(HOME)
                    }else
                    {
                        LastDeathCoords = HOME
                        print("LastDeathCoods set to HOME")
                    }
                }else
                {
                    print("HOME not set. do !sethome first")
                }
                
            break
            case "!tp",
            case "!teleport",
                if(Argument != "jailer")
                {
                    local PlayerName = findPlayerByName(Argument)
                    local POS = PlayerName:pos()
                    if(PlayerName:isValid() & POS:isInWorld())
                    {
                        # try to spawn above the player
                        # if it's not possible , spawn behind (ground)
                        # else inside their body (croush to unstuck)
                        Above = PlayerName:pos() + vec(0,0,PlayerName:height()) + vec(0,0,owner():height() + 10)
                        Behind = PlayerName:shootPos()-PlayerName:eye()*40
                        Behind_Down = rangerOffset(Behind-vec(0,0,5),Behind-vec(0,0,1000000))
                        if(Above:isInWorld())
                        {
                            POS = Above
                        }
                        elseif(Behind_Down:pos():isInWorld()){
                            POS = Behind_Down:pos()
                        }else{
                            POS = PlayerName:pos() + vec(0,0,30)
                        }
                        
                        if(!owner():isAlive())
                        {
                            LastDeathCoords = POS
                            print("LastDeathCoods set to PLAYER")
                        }else
                        {
                            goto(POS)
                        }
                    }
                }else
                {
                    if(Jailer:isValid())
                    {
                        DelayAfterTP = 1
                        if(!owner():isAlive())
                        {
                            LastDeathCoords = Jailer:pos()+vec(0,0,-10)
                            print("LastDeathCoods set to JAILER")
                        }else
                        {
                            goto(Jailer:pos()+vec(0,0,-10))
                        }
                    }else{print("Jailer isn't valid")}
                }
            break
            case "!surface",
            case "!up",
                if(Sky != vec(0,0,0))
                {
                    Sky[1,number] = owner():pos():x()
                    Sky[2,number] = owner():pos():y()
                    Terrain = rangerOffset(Sky,Sky-vec(0,0,1000000))
                    
                    goto(Terrain:pos())
                }else
                {
                    findSky()
                }
            break
            case "!chip",
                if(!owner():isAlive())
                {            
                    LastDeathCoords = entity():pos()
                    print("LastDeathCoods set to CHIP")
                }else
                {
                    goto(entity():pos())
                }
            break
            case "!chiphere",
                entity():setPos(owner():pos()+vec(0,0,1))
            break
            case "!look",
            case "!lk",
                InitialVector = owner():pos()+vec(0,0,owner():height())
                DestinationVector = owner():aimPos()
                
                DirectionVector = DestinationVector - InitialVector
                Magnitude = sqrt(DirectionVector:x()^2 + DirectionVector:y()^2 + DirectionVector:z()^2)
                NormalizedDirectionVector = DirectionVector / Magnitude
                
                BackwardVector = NormalizedDirectionVector * owner():height() # using the player's height should avoid being stuck, increase if needed
                
                NewPositionVector = DestinationVector - BackwardVector
                if(NewPositionVector:isInWorld())
                {
                    goto(NewPositionVector)
                }
            break
            case "!lg",
                rangerHitEntities(1)
                rangerHitWater(0)
                const Ranger = rangerOffset(100000,owner():aimPos(),vec(0,0,-1))
                local Pos = Ranger:pos()
                local Unit = 0
                while(Pos:isInWorld())
                {
                    Unit -= 2
                    Pos = Pos+vec(0,0,Unit)
                }
                goto(Pos)
            break
            case "!forward",
            case "!fw",
                Forward = owner():shootPos()+owner():eye()*Argument:toNumber()
                
                if(1 == 1) #Forward:isInWorld()
                {
                    goto(Forward)
                    
                }else{print("Can't reach that!")}
            break
            case "!traverse",
            case "!tr",
            case "!bypass",
            case "!by",
                InitialVector = owner():pos()+vec(0,0,owner():height())
                DestinationVector = owner():aimPos()
                DirectionVector = InitialVector - DestinationVector
                
                Magnitude = sqrt(DirectionVector:x()^2 + DirectionVector:y()^2 + DirectionVector:z()^2)
                NormalizedDirectionVector = DirectionVector / Magnitude
                
                
                for(I=100,10000,10)
                {
                    BackwardVector = NormalizedDirectionVector * (I + owner():height())
                    NewPositionVector = DestinationVector - BackwardVector
                    
                    if(NewPositionVector:isInWorld())
                    {
                        
                        # Laser down to find potential ground
                        local Ranger = rangerOffset(100,NewPositionVector,vec(0,0,-1))
                        const Ground = Ranger:pos()
                        
                        goto(Ground)
                        break
                    }
                }
            break
            case "!sky",
                findSky()
            break
            case "!killer",
                if(Killer:isValid())
                {
                    LastDeathCoords = Killer:pos()
                    print("LastDeathCoords set to Killer")
                    Killer = noentity()
                }else{print("Invalid killer")}
            break
            case "!perma",
                local AAA = findPlayerByName(Argument)
                if(AAA:isValid())
                {
                    Perma = AAA
                    print("Perma respawn set to: "+Perma:name())
                }else{print("Invalid."), Perma = noentity()}
            break
            case "!void",
                Void = Voids[map(),vector]
                if(Void != vec(0,0,0))
                {
                    if(!owner():isAlive())
                    {
                        LastDeathCoords = Void
                        VoidRespawn = 1
                        JailerRespawn = 0
                        print("LastDeathCoods set to Void")
                    }else
                    {
                        goto(Void)
                    }
                }else{ print("Void not set for current map!")}
            break
            case "!egg",
            case "!ee",
            case "!room",
                Room = EERooms[map(),vector]
                if(Room != vec(0,0,0))
                {
                    if(!owner():isAlive())
                    {
                        LastDeathCoords = Room
                        VoidRespawn = 1
                        JailerRespawn = 0
                        print("LastDeathCoods set to Room")
                    }else
                    {
                        goto(Room)
                    }
                }else{ print("Easter Egg Room not set for current map!")}
            break
            case "!reverse",
                local PlayerName = findPlayerByName(Argument)
                
                # Failsafe for players with short names
                if(!PlayerName:isValid())
                {
                    const Players_ = players()
                    for(I=1,Players_:count())
                    {
                        if(Players_[I,entity]:name() == Argument){ PlayerName = Players_[I,entity]}
                    }
                }
                if(PlayerName:isValid() & Argument != "")
                {
                    ReversedPlayer = PlayerName
                    LastDeathCoords = ReversedPlayer:pos()+vec(0,0,40)
                    Reverse = 1
                    print("Reverse ON --> ",PlayerName:name())
                    Jailer = noentity()
                    JailerRespawn = 0
                }else
                {
                    Reverse = 0
                    print("Reverse OFF")
                    
                    owner():setMaterial("")
                    owner():weapon():setMaterial("")
                }
            break
            
        }
        
        
    }
}

event playerDeath(Victim:entity, Inflictor:entity, Attacker:entity) {
    
    const Me = owner()
    if(Victim == Me & Me:pos():isInWorld()) #
    {
        concmd("stopsound")
        ID = 0
        
        LastDeathCoords = Victim:pos()
        People = players()
        
        
        if(Inflictor:isValid()){Killer = Inflictor}
        elseif(Attacker:isValid()){Killer = Attacker}
        else{Killer = noentity()}
        
        
        # ---------------------------------
        # DETECT SPAWN KILLERS
        # ---------------------------------
        
        # Any type of spawn killing
        LastDeaths:pushVector(LastDeathCoords)
        if(LastDeaths:count() > 3)
        {
            LastDeaths:removeVector(1)
            
            local DistDeaths = LastDeaths[1,vector]:distance(LastDeaths[2,vector])
            local DistDeaths2 = LastDeaths[2,vector]:distance(LastDeaths[3,vector])
            local DistDeaths3 = LastDeaths[2,vector]:distance(LastDeaths[3,vector])
            
            
            # Another type of spawn killing
            if(DistDeaths < 200 & DistDeaths2 < 200 & DistDeaths3 < 200 & Inflictor:type() != "gmod_wire_simple_explosive")
            {
                Void = Voids[map(),vector]
                if(Void != vec(0,0,0))
                {
                    # Check if the spawn killing has been happening near the chip
                    
                    if(LastDeaths[3,vector]:distance(entity():pos()) < 200)
                    {
                        Void = Voids[map(),vector]
                        LastDeathCoords = Void
                        VoidRespawn = 1
                        
                    }else{
                        LastDeathCoords = entity():pos()
                    }
                    
                    print("Detected multiple deaths on the same spot.")
                }
            }
        }
        
        # Specific to Wire Jailers
        if(Inflictor:type() == "gmod_wire_simple_explosive" && !JailerRespawn)
        {
            if(findCanQuery())
            {
                findIncludeClass("gmod_wire_teleporter")
                findInSphere(Victim:pos(),15)
                
                Array = findToArray()
                
                # gmod_wire_simple_explosive   9.4170999526978
                # gmod_wire_teleporter         2.9998199939728
    
                for(I=1,Array:count())
                {
                    Jailer = Array[I,entity]:owner()
                    if(!NoVoids)
                    {
                        Void = Voids[map(),vector]
                        LastDeathCoords = Void
                        VoidRespawn = 1
                    }else
                    {
                        # Find random void
                        ReversedPlayer = Jailer
                        LastDeathCoords = ReversedPlayer:pos()+vec(0,0,40)
                        Reverse = 1
                        JailerRespawn = 0
                    }
                    
                    print("You have been targeted by a Wire Jailer: " + Jailer:name() + ". Chip will try to respawn you in a safe zone.")
                }
            }
        }
        
        # Determine if what killed you is a player or a prop
        local TrueAttacker = ""
        if(Attacker:isPlayer())
        {
            TrueAttacker = Attacker:name()
        }else
        {
            TrueAttacker = Inflictor:owner():name()
        }
        
        # Some entities don't have owners, we can still try to get them via getConnectedEntities
        if(TrueAttacker == "")
        {
            local ConnectedProps = Inflictor:getConnectedEntities()
            local ConnectedProp  = noentity()
            for(I=1,ConnectedProps:count())
            {
                if(ConnectedProps[I,entity] == Inflictor | ConnectedProps[I,entity]:isPlayer()){ continue }
                if(!ConnectedProps[I,entity]:isOnFire() | !ConnectedProps[I,entity]:isValid()){ continue }
                
                ConnectedProp = ConnectedProps[I,entity]
            }
            TrueAttacker = ConnectedProp:owner():name()
        }
        
        # Print instructions        
        if(LastDeathCoords == Victim:pos())
        {
            print("You got killed by:   "+TrueAttacker + "    --  (press --> (right) to change respawn point)")
            #print("Inflictor: "+Inflictor:name()+ "   --   Attacker: "+Attacker:name() + "      (press --> (right) to change respawn point)")
        }
    
        
        Debug = ".\n"+
        "Killer: "+Killer:name() 
    }
}

event keyPressed(Player:entity, Key:string, Down:number, KeyBind:string) {
    if(Player != owner()){ exit() }
    
    if(Down & Key == "right" & !owner():isAlive())
    {
        ID++
    }
    
    if(Down & Key == "left" & !owner():isAlive())
    {
        ID--
    }
    
    #if(Key == "f" & Down){ Increaser++ , print(Increaser)}
    if(Down & !owner():isAlive())
    {
        if(ID < 0){ID = 1}
        if(ID > People:count()+1){ID--}
        if(ID == People:count()+1){
            LastDeathCoords = owner():pos()
            print("Your location")        
        }else
        {
            if(ID != 0)
            {
                # try to spawn above the player
                # if it's not possible , spawn behind (ground)
                # else inside their body (croush to unstuck)
                Above = People[ID,entity]:pos() + vec(0,0,People[ID,entity]:height()) + vec(0,0,owner():height() + 10)
                Behind = People[ID,entity]:shootPos()-People[ID,entity]:eye()*40
                Behind_Down = rangerOffset(Behind-vec(0,0,5),Behind-vec(0,0,1000000))
                if(Above:isInWorld())
                {
                    LastDeathCoords = Above
                }
                elseif(Behind_Down:pos():isInWorld()){
                    LastDeathCoords = Behind_Down:pos()
                }else{
                    LastDeathCoords = People[ID,entity]:pos() + vec(0,0,30)
                }
                
                
                print(People[ID,entity]:name() + " is your respawn target")
            }
        }
    }
}

event playerSpawn(Player:entity) {
    
    if(Player == owner())
    {
        
        if(!Reverse)
        {
            owner():setMaterial("")
            owner():weapon():setMaterial("")
        }else{
            owner():setMaterial("models/effects/vol_light001")
            owner():weapon():setMaterial("models/effects/vol_light001")
        }
        
        if(Perma:isValid())
        {
            LastDeathCoords = Perma:pos()
            
            Above = Perma:pos() + vec(0,0,Perma:height()) + vec(0,0,owner():height() + 10)
            Behind = Perma:shootPos()-Perma:eye()*40
            Behind_Down = rangerOffset(Behind-vec(0,0,5),Behind-vec(0,0,1000000))
            if(Behind_Down:pos():isInWorld())
            {
                LastDeathCoords = Behind_Down:pos()
            }
            elseif(Above:isInWorld()){
                LastDeathCoords = Above
                
            }else{
                LastDeathCoords = Perma:pos() + vec(0,0,30)
            }
        }
        
        if(LastDeathCoords != vec(0,0,0) & (LastDeathCoords:isInWorld() || VoidRespawn) & !JailerRespawn)
        {
            CHAIR:setPos(LastDeathCoords)
            CHAIR:use()
            
            WaitChairTP = 1
            VoidRespawn = 0
        }
        
        if(LastDeathCoords != vec(0,0,0) & (LastDeathCoords:isInWorld() || JailerRespawn) & !VoidRespawn)
        {
            CHAIR:setPos(Jailer:pos())
            CHAIR:use()

            WaitChairTP = 1
            VoidRespawn = 0
        }
        
    }
}

event tick()
{
    if(owner():isFlashlightOn()) { goto(HOME) }
    
    if(DELAY > 0){DELAY--}
    if(DelayAfterTP > 0){DelayAfterTP--}

    if(WaitChairTP & DELAY == 0 & !Reverse)
    {
        if(CHAIR:driver():isValid() & CHAIR:driver() == owner() & owner():pos():distance(CHAIR:pos()) < 23 & CHAIR:isValid())
        {
            #CHAIR:ejectPod()
            
            CHAIR:propDelete()
            LastDeathCoords = vec(0,0,0)
            WaitChairTP = 0
            print("Teleported")
        
            
        }
    }
    
    if(Reverse & owner():isAlive())
    {
        CHAIR:setAng(ang(180,0,0))
        
        local Pos = ReversedPlayer:pos()+vec(0,0,ReverseHeight)
        
        if(Pos:isInWorld())
        {
            CHAIR:setPos(Pos)
            
            CHAIR:propNotSolid(1)
            CHAIR:setAlpha(0)
        }
    }
      
    
    # In case the chair gets deleted
    if(!CHAIR:isValid() & propCanCreate())
    {
        CHAIR = seatSpawn("models/nova/chair_plastic01.mdl", entity():pos()+vec(0,0,30),ang(0,0,0),1)
        CHAIR:propNotSolid(1)
        CHAIR:propFreeze(1)
        CHAIR:setAlpha(0)    
    }
    
    

  
}
