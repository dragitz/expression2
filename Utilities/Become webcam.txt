event chat(Player:entity, Message:string, Team:number) {
    
    if(Player == owner() && Message[1] == "!")
    {
        hideChat(1)
        Exploded = Message:explode(" ")
        
        Command = Exploded[1,string]        
        
        switch(Command)
        {
            case "!delete",
                selfDestructAll()
                CHAIR:propBreak()
            break
        }
    }
}

{
    #if(Me:aimEntity() == Obj){ Obj:propNotSolid(1) }else{ Obj:propNotSolid(0) }
    Forward = Player:shootPos()+Player:eye()*130  # normal

        CHAIR:setPos(Forward)

    }

event playerSpawn(Player:entity) {
    if(Player == owner())
    {
        owner():setMaterial("models/effects/vol_light001") owner():weapon():setMaterial("models/effects/vol_light001")
        CHAIR:use()
    }
}

event playerDisconnected(DC:entity) {
    if(DC == Player)
    {
        CHAIR:propDelete()
        selfDestruct()
    }
}
event removed(N:number)
{
    owner():setMaterial("") owner():weapon():setMaterial("")
}
