event tick()
{

    #debug
    #Prop:setPos(owner():pos()+vec(0,0,owner():height()+10*owner():isOnGround()))
    #Prop:setAng(ang(180,owner():angles():yaw()+90,0))

    
    CHAIR:setAng(ang(180,0,0))
    
    if(AtRisk & CHAIR:driver() != owner())
    {
        
        #if(LastKiller:isCrouch()){ SetHeight = HEIGHT_CROUCH }
        
        const SetHeight = LastKiller:attachmentPos("eyes"):z() - LastKiller:pos():z() - HEIGHT
        
        Prop:setPos(LastKiller:pos()+vec(0,0,SetHeight))
        Prop:setAng(ang(PITCH,LastKiller:angles():yaw()+90,EXTRA))
    }

    if(!AtRisk & CHAIR:driver():isPlayer() & RespawnAmount >= MinRespawnAmount)
    {
        #CHAIR:printDriver("Congrats! ")
        CHAIR:printDriver(_HUD_PRINTCENTER,
            LastRevenger:name() + " got gotted by\n\n
            Anti Killer be killed+"
        )
        CHAIR:setPos(OriginalPosition)
        CHAIR:ejectPod()

        RespawnAmount = 0
    }

        # Always TP the invisible chair under the revenger's feet
        CHAIR:setPos(LastRevenger:pos() + vec(0, 0, 0))
        
        # Remove invalid turrets
        local Keys = Turrets:keys()
        for(I=1,Keys:count())
        {
            const ID = Keys[I,number]
            if(!entity(ID):isValid()){ Turrets:unset(ID), continue }
        }
        
        # Unset turret owners if there are no owned turrets by them
        TotalTurrets = 0
        People = table()
        Keys = Turrets:keys()
        local TurretList = array()
        for(I=1,Keys:count())
        {
            const ID = Keys[I,number]
            const Entity = entity(ID)
            const Steam = Entity:owner():steamID64()
            People[Steam,number] = 1
            
            TurretList:pushEntity(Entity)
            TotalTurrets += 1
        }
        
        
        # simple reset
        if(!TotalTurrets){ AtRisk = 0 }
        else
        {
            # advanced reset
            if(AtRisk & findCanQuery())
            {
                local Found = 0
                for(I=1,TurretList:count())
                {
                    const Turr = TurretList[I,entity]
                    const D = Turr:pos():distance(LastKiller:pos())
                    if(D < 130){Found += 1}
                }
                if(!Found){ AtRisk -= 1}
            }
        }
        

        if(AtRisk)
        {
            CachedName = "Anti Killer be killed+ \n\n Protecting: "+LastKiller:name()+"\n\n"
            CachedName += "--> Press E (or your USE button) <--- \n\n\n"
            CachedName += LastRevenger:name()+" is trying to kill you!\n"
        }else{

            CachedName = "Anti Killer be killed+"
        }
        if(CachedName != entity():getName()){ entity():setName(CachedName) }
    
}

    if(!Data[Entity:type(),number]){ exit() }    
    if(!Data[Entity:getName(),number]){ exit() }
    
    # Add turret to list of turrets
    local Turret = noentity()
    const Ents = Entity:getConnectedEntities("")
    for(I=1,Ents:count())
    {
        if(Ents[I,entity]:type() == "gmod_wire_turret"){ Turret = Ents[I,entity], break}
    }
    
    # not an auto revenge e2 but still has the same name?
    if(!Turret:isValid())
    {
        print("Invalid turret? owner: "+Entity:owner():name())
        exit()
    }
    
    # Create slot
    # turret is valid, add it to the group of tracked turrets
    if(!Turrets[Turret:owner():steamID64(),table])
    {
        # entity id may not be the best, but tick will remove it from the list, so should be fine
        Turrets[Turret:id(),number] = 1
        People[Turret:owner():steamID64(),number] = 1
    }
    
    
    People[Entity:owner():steamID64(),table]:pushEntity(Turret)
    #const Keys = People[Entity:owner():steamID64(),table]:keys()
    
    print("Player: " + Entity:owner():name() + " spawned an auto revenge")

}

event playerDeath(Victim:entity, Inflictor:entity, Attacker:entity) {
    if(!People[Victim:steamID64(),number]){ exit() }
    
    # not protecting people who have the revenge turret
    if(People[Attacker:steamID64(),number]){ exit() }
    
    
    #print(Inflictor, Inflictor:owner())
    if(Inflictor:type() == "prop_physics" & Inflictor:owner() == Victim & CHAIR:driver():isPlayer())
    { 
        RespawnAmount += 1
        CHAIR:printDriver(_HUD_PRINTCENTER,"Please wait.. \n\nOne more death till you are safe again!\n\n If you feel safe, you can exit the vehicle!")
    }

    # i don't want to code something that detects kills based on props, only players
    if(!Attacker:isPlayer()){ exit() }
    
    LastKiller = Attacker
    LastRevenger = Victim
    
    # Prop that protects the player from being revenge killed
    
    const SetHeight = LastKiller:attachmentPos("eyes"):z() - LastKiller:pos():z() - HEIGHT
    Prop:setPos(LastKiller:pos()+vec(0,0,SetHeight))
    Prop:setAng(ang(PITCH,LastKiller:angles():yaw()+90,EXTRA))
    
    AtRisk = 3
    
    OriginalPosition = LastKiller:pos()
}
