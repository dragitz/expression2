@name Anti Wire Jailers 3.1
@persist [Blacklisted_E2 Blacklisted_WIRE Blacklisted_PROPS LinkedProps LinkedPropsExplosives]:table
@persist LastScanTime [PropQueue CHAIR]:entity TRIGGER_ENTITY:entity CHAIR_TICK



#[
    Changes in 3.1:
        - Teleport the player inside the jail (and also prevent from activating it)
    
    The concept is pretty simple:
    
    1) One of the most commonly used jails to annoy people has a screen
    2) Upon spawn, we place a plate in front of it
    3) Jail becomes useless
    4) ????
    5) Profit!
    
    If the jail is removed, the plate associated also gets removed
    
    Note: does not work on already spawned jails
    
]#

if(first() | duped())
{
    propSpawnUndo(0)
    
    entity():propMakePersistent(1)
    # Setup
    Blacklisted_E2 = table(
        "Jail with EGP" = 1
    )
    
    Blacklisted_WIRE = table(
        "gmod_wire_egp" = 1,
        "gmod_wire_simple_explosive" = 1
        #"gmod_wire_teleporter" = 1
    )
    
    Blacklisted_PROPS = table(
        "gmod_wire_expression2" = 1
    )
    
    LinkedProps = table()
    LinkedPropsExplosives = table()
    
    CHAIR_TICK = 0
    TRIGGER_ENTITY = noentity()
    PropQueue = noentity()
    
    LastScanTime = systime()
    
    
    function array findJailDupe(E2:entity)
    {
        # Find attached Screen
        const Array = E2:getConnectedEntities()
        
        local PropsArray = array()
        
        for(I=1,Array:count())
        {
            local Prop = Array[I,entity]
            if(!Blacklisted_WIRE[Prop:type(),number]){ continue }
            
            # Prop has been found
            if(Prop:type() == "gmod_wire_egp"){              PropsArray[1,entity] = Prop }
            if(Prop:type() == "gmod_wire_simple_explosive"){ PropsArray[2,entity] = Prop }
        }
        
        return PropsArray
    }
    
    CHAIR = seatSpawn("models/nova/chair_plastic01.mdl", entity():pos()+vec(0,0,30),ang(0,0,0),1)
    CHAIR:setAlpha(0)
    CHAIR:setCollisionGroup(_COLLISION_GROUP_PASSABLE_DOOR)
    CHAIR:propDraw(0)
    
    
    if(!CHAIR:isValid()){ print("Chair did not spawn, please respawn the chip")}
    
}

event keyPressed(Player:entity, Key:string, Down:number, KeyBind:string) {
    
    # prevent chip error
    #if(!Blacklisted[Player:name(),number]){ exit() }
    if(KeyBind == "use" & Down)
    {
        # Check if we are looking at a prop in LinkedProps
        const Prop = Player:aimEntity()
        if(Prop)
        {
            if(LinkedPropsExplosives[Prop:id(),entity]:isValid())
            {
                TRIGGER_ENTITY = LinkedPropsExplosives[Prop:id(),entity]
                
                Forward = Player:shootPos()+Player:eye()*(-1.3)
                CHAIR:setPos(Forward)
        
                Pitch = Player:eyeAngles():pitch() + 90
                Yaw = Player:eyeAngles():yaw()
                Roll = Player:eyeAngles():roll()
                CHAIR:setAng(ang(Pitch,Yaw,Roll))
            }
        }
    }
}

event playerEnteredVehicle(Player:entity, Vehicle:entity) {
    if(Vehicle != CHAIR){ exit() }
    
    print(Player:name(), " got gotted")
}


event entityCreated(Entity:entity) {
    
    # Prevent chip crash
    if(!Blacklisted_PROPS[Entity:type(),number]){ exit() }
    
    # Only works on dupes
    local Name = Entity:getName()
    
    # Ensure it is a known e2 jail
    if(!Blacklisted_E2[Name,number]){exit()}
    
    # Get attached screen
    local ReturnArray = findJailDupe(Entity)
    
    local Screen = ReturnArray[1,entity]
    local Explosive = ReturnArray[2,entity]
    
    # Check if wire prop exists or not
    if(!Screen:isValid()){ print("Invalid Screen"), exit()}
    if(!Explosive:isValid()){ print("Invalid Explosive"), exit()}
    
    # Screen is valid
    print(Entity:owner():name() + " spawned a jail")
    
    local CounterProp2 = PropQueue
    PropQueue = noentity()
    
    # If prop isn't valid
    if(!CounterProp2:isValid())
    {
        print("Something went wrong: CounterProp2 is invalid (check prop queue)")
        exit()
    }
    
    # Prop is valid
    CounterProp2:setAlpha(0)
    CounterProp2:propDraw(0)
    CounterProp2:setCollisionGroup(_COLLISION_GROUP_PASSABLE_DOOR)
    
    
    # Link Screen with CounterProp
    LinkedProps[CounterProp2:id(),entity] = Screen
    LinkedPropsExplosives[CounterProp2:id(),entity] = Explosive
    
}

event tick()
{
    # Scan every x ms
    if(systime() - LastScanTime > 0.05 & perf())
    {
        LastScanTime = systime()
        
        Keys = LinkedProps:keys()
        #print(Keys)
        for(I=1,Keys:count())
        {
            # Key holds the entity id, which can be used to retrieve the associated Screen id
            local Key = Keys[I,number]
            
            local ExplosiveEnt = LinkedProps[Key,entity]
            local CounterProp = entity(Key)
            
            # Check if Jails has been removed
            if(!ExplosiveEnt:isValid()){ CounterProp:propDelete(), LinkedProps:unset(Key), continue }
            
            
            # Keep the egp blocker in place (in case they moved the jail)
            local Pos = ExplosiveEnt:pos()+ExplosiveEnt:up()/9
            
            CounterProp:setPos(Pos)
            CounterProp:setAng(ExplosiveEnt:angles())
        }
        
        # Respawn items in the queue
        if(!PropQueue:isValid() & propCanCreate())
        {
            PropQueue = propSpawn("models/hunter/plates/plate1x1.mdl",entity():pos()+vec(0,0,30),ang(0,0,0),1)
            PropQueue:propDraw(0)
        }
        
        # Check if there's a driver in our chair
        if(CHAIR:driver():isValid())
        {
            CHAIR:setAng(ang(0,0,0))
            CHAIR:setPos(TRIGGER_ENTITY:pos())
            
            if(CHAIR_TICK > 1)
            {
                CHAIR:propDelete()
                CHAIR_TICK = 0
            }
            
            CHAIR_TICK += 1
        }
        # Check if CHAIR has been deleted
        if(!CHAIR:isValid())
        {
            CHAIR = seatSpawn("models/nova/chair_plastic01.mdl", entity():pos()+vec(0,0,30),ang(0,0,0),1)
            CHAIR:setAlpha(0)
            CHAIR:setCollisionGroup(_COLLISION_GROUP_PASSABLE_DOOR)
            CHAIR:propDraw(0)
        }
    }
}

# In case it gets halted, fully reset the chip
event removed(Resetting:number) {
    propDeleteAll()
    reset()
}
