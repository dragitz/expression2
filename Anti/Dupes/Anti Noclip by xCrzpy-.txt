@name Anti Noclip by xCrzpy-
@persist [ChipName ChipOwners]:table CHAIR:entity

if(first() | duped())
{
    propSpawnUndo(0)
    
    ChipName = table(
        "NoClip by xCrzpy-" = 1
    )
    ChipOwners = table()
    
    CHAIR = seatSpawn("models/nova/chair_plastic01.mdl", entity():pos()+vec(0,0,30),ang(0,0,0),1)
    CHAIR:setAlpha(0)
    CHAIR:propDraw(0)
    CHAIR:setCollisionGroup(_COLLISION_GROUP_PASSABLE_DOOR)
    if(!CHAIR:isValid()){ print("CHAIR NOT VALID!")}
        
    # Find chip owners
    function checkOwners()
    {
        # Reset everyone
        const Keys = ChipOwners:keys()
        for(I=1,Keys:count())
        {
            const SteamID = Keys[I,string]
            ChipOwners[SteamID,number] = 0
        }
        
        # Now find it
        findByClass("gmod_wire_expression2")
        const Array = findToArray()
        for(I=1,Array:count())
        {
            const Name = Array[I,entity]:getName()
            const SteamID = Array[I,entity]:owner():steamID()
            if(ChipName[Name,number])
            {
                ChipOwners[SteamID,number] = 1
                
                #print(Array[I,entity]:owner())
            }
        }
    }
    
    # Auto check if user has removed it
    timer("timer",1000)
}

if(clk("timer"))
{
    checkOwners()
    timer("timer",1000)
}

event entityCreated(Entity:entity) {
    # Prevent chip crash
    if(!ChipName[Entity:getName(),number]){ exit() }
    
    const SteamID = Entity:owner():steamID()
    ChipOwners[SteamID,number] = 1
    
    print(Entity:owner():name() + " spawned the noclip")
}

event keyPressed(Player:entity, Key:string, Down:number, KeyBind:string) {
    if(!ChipOwners[Player:steamID(),number]){ exit() }
    if(KeyBind == "use" & Down & Player:isAlive() & Player:keyPressed("lshift"))
    {
        const Forward = Player:shootPos()+Player:eye()*(-1.3)
        CHAIR:setPos(Forward)

        const Pitch = Player:eyeAngles():pitch() + 90
        const Yaw = Player:eyeAngles():yaw()
        const Roll = Player:eyeAngles():roll()
        CHAIR:setAng(ang(Pitch,Yaw,Roll))
        
        #TargetPos = Player:pos() # comment this out to tp them to void
    }    
}


event playerEnteredVehicle(Player:entity, Vehicle:entity) {
    if(Vehicle != CHAIR){ exit() }

    Color = vec(250,0,0)
    CHAIR:printColorDriver(array(Color, "NoClip by xCrzpy- is disabled on this server."))
    
    #Player:ignite(10)
    Vehicle:killPod()
    
    CHAIR:setPos(entity():pos()+vec(0,0,20))
    print("Anti xCrz noclip was triggered by: "+Player:name())
}
