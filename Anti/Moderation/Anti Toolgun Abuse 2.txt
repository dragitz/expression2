    if(!Data[KeyBind,number]){ exit() }
    #if(!Data[Player:tool(),number]){ exit() }
    if(!Data[Player:weapon():type(),number]){ exit() }
    
    # Extra checks
    if(!Down){ exit() }
    if(Player == owner()){ exit() }
    
    # Forced blacklist
    if(People[Player:steamID(),number])
    {
        Forward = Player:shootPos()+Player:eye()* 70
        CHAIR:setPos(Forward)
    
        Pitch = Player:eyeAngles():pitch() + 90
        Yaw = Player:eyeAngles():yaw()
        Roll = Player:eyeAngles():roll()
        CHAIR:setAng(ang(Pitch,Yaw,Roll))
        
        Cooldown = 3
        print(Player:name() + " tried to use dupe tool")
        exit()        
    }
    
    # Check if the player is in cooldown
    if(!Players[Player:steamID(),number])
    {
        Players[Player:steamID(),number] = systime() + Delay
        exit()
    }
    
    if(systime() < Players[Player:steamID(),number])
    {
        Me = Player
        Forward = Me:shootPos()+Me:eye()* 70
        CHAIR:setPos(Forward)

        Yaw = Me:eyeAngles():yaw()
        Roll = Me:eyeAngles():roll()
        CHAIR:setAng(ang(Pitch,Yaw,Roll))
        
        Cooldown = 3
        exit()
        
    }
    Players[Player:steamID(),number] = systime() + Delay
}
