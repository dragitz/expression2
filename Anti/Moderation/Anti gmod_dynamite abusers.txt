event entityCreated(Entity:entity) {
    
    # Prevent chip crashes
    if( !Types[Entity:type(),number] ) { exit() }
    if( !BlackListedProps[Entity:model(), number] ){ exit() }
    
    
    Player  = Entity:owner()
    SteamID = Player:steamID()            

    # Avoid further crashes
    if( Exploders[SteamID,number] > 100){ exit() }

    # Assign score
    if(!Exploders:exists(SteamID))
    {
        Exploders[SteamID,number] = 1
    }
    else
    {
        if(Exploders[SteamID,number] < Threshold)
        {
            Exploders[SteamID,number] = Exploders[SteamID,number] + 1
        }
    }
}
